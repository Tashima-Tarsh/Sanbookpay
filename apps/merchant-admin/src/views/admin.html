import fetch from 'node-fetch';

async function createSanbookPayment() {
  const response = await fetch('http://localhost:6100/api/v1/payments', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-Merchant-Key': process.env.SANBOOK_SECRET_KEY,
      'Idempotency-Key': 'ORD-1001'
    },
    body: JSON.stringify({
      orderId: 'ORD-1001',
      amount: '499.00',
      currency: 'INR',
      customer: {
        name: 'Mohit Sharma',
        phone: '9876543210',
        email: 'mohit@example.com'
      }
    })
  });

  const data = await response.json();
  console.log('Customer Auto-Generated:', data.customerId);
  console.log('Masked Phone:', data.customer.phone); // +91 98*** ***10
  console.log('Checkout URL:', data.checkoutUrl);
  return data;
}
